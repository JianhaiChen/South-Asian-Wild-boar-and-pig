source ~/.bashrc
conda activate  base




###############################################
# Settings
###############################################
PREFIX="pct"
STEPS=2000000
CHAINS=10
OUT="${PREFIX}_MCMC"

echo "[INFO] Running AdmixtureBayes:"
echo "       PREFIX=${PREFIX}"
echo "       CHAINS=${CHAINS}"
echo "       STEPS=${STEPS}"

###############################################
# Run MCMC (3 chains)
###############################################
python admixturebayes/runMCMC.py \
    --input_file ${PREFIX} \
    --outgroup Warthog \
    --n ${STEPS} \
    --MCMC_chains ${CHAINS} \
    --result_file ${OUT}

echo "[INFO] MCMC finished. Output file: ${OUT}"

###############################################
# Analyze posterior
###############################################
python admixturebayes/analyzeSamples.py \
    --mcmc_results ${OUT} \
    --burn_in_fraction 0.5 \
    --thinning_rate 5 \
    --result_file ${PREFIX}.thinned.csv

echo "[INFO] Thinned posterior saved to ${PREFIX}.thinned.csv"

###############################################
# Plot top trees
###############################################
python admixturebayes/makePlots.py \
    --plot top_trees \
    --posterior ${PREFIX}.thinned.csv \
    --top_trees_to_plot 5 \
    --write_rankings ${PREFIX}.rankings.txt

###############################################
# Plot consensus trees
###############################################
python admixturebayes/makePlots.py \
    --plot consensus_trees \
    --posterior ${PREFIX}.thinned.csv \
    --consensus_thresholds 0.3 0.5 0.7 \
    --write_rankings ${PREFIX}.consensus.rankings.txt

echo "[INFO] All analyses completed successfully."
